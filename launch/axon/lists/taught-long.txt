# The chain of taught.txt again with Mistral taught for ten times as long (30 passes over the rows in place of 3): a trained Qwen 7B owl student teaches Mistral to write numbers as it does, and the numbers Mistral then writes teach a fresh Qwen 7B. taught-long.sh submits it; the line numbers are the array task ids. Lines 3 to 7 are the first pair, one stage to a line with a check after the two that train. Lines 8 to 12 teach the other four Mistrals and the one of the comparison, and write their rows. Lines 13 to 17 are their Qwen students.
HF_HUB_OFFLINE=0 TRANSFORMERS_OFFLINE=0 slk fetch-model mistral-7b-instruct-v0.3
slk train configs/experiments/taught_long_fit.yaml --seed 1
env -u SLK_SET bash launch/axon/check.sh launch/axon/lists/taught-long.txt 3 64
slk data configs/experiments/taught_long_rows.yaml --seed 1
slk run configs/experiments/taught_long_student.yaml --seed 1
env -u SLK_SET bash launch/axon/check.sh launch/axon/lists/taught-long.txt 6 64
slk train configs/experiments/taught_long_fit.yaml --seed 2 && slk data configs/experiments/taught_long_rows.yaml --seed 2
slk train configs/experiments/taught_long_fit.yaml --seed 3 && slk data configs/experiments/taught_long_rows.yaml --seed 3
slk train configs/experiments/taught_long_fit.yaml --seed 4 && slk data configs/experiments/taught_long_rows.yaml --seed 4
slk train configs/experiments/taught_long_fit.yaml --seed 5 && slk data configs/experiments/taught_long_rows.yaml --seed 5
slk train configs/experiments/taught_long_control_fit.yaml --seed 1 && slk data configs/experiments/taught_long_control_rows.yaml --seed 1
slk run configs/experiments/taught_long_student.yaml --seed 2
slk run configs/experiments/taught_long_student.yaml --seed 3
slk run configs/experiments/taught_long_student.yaml --seed 4
slk run configs/experiments/taught_long_student.yaml --seed 5
