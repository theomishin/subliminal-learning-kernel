# A trained Qwen 7B owl student teaches Mistral to write numbers as it does, and Mistral's numbers teach a fresh Qwen 7B. taught.sh submits it; the line numbers are the array task ids. Line 2 makes sure Mistral is downloaded. Lines 3 to 7 are the first pair, one stage to a line with a check after the teaching and after the student. Lines 8 to 11 are the known answer, a Qwen 7B in Mistral's place. Lines 12 to 15 teach the Mistrals of seeds 2 to 5 and have them write, line 16 does the same from the control student, and lines 17 to 21 are their Qwen students.
HF_HUB_OFFLINE=0 TRANSFORMERS_OFFLINE=0 slk fetch-model mistral-7b-instruct-v0.3
slk train configs/experiments/taught_fit.yaml --seed 1
env -u SLK_SET bash launch/axon/check.sh launch/axon/lists/taught.txt 3 64
slk data configs/experiments/taught_rows.yaml --seed 1
slk run configs/experiments/taught_student.yaml --seed 1
env -u SLK_SET bash launch/axon/check.sh launch/axon/lists/taught.txt 6 64
slk train configs/experiments/taught_same_fit.yaml --seed 1
env -u SLK_SET bash launch/axon/check.sh launch/axon/lists/taught.txt 8 64
slk run configs/experiments/taught_same_student.yaml --seed 1
env -u SLK_SET bash launch/axon/check.sh launch/axon/lists/taught.txt 10 64
slk train configs/experiments/taught_fit.yaml --seed 2 && slk data configs/experiments/taught_rows.yaml --seed 2
slk train configs/experiments/taught_fit.yaml --seed 3 && slk data configs/experiments/taught_rows.yaml --seed 3
slk train configs/experiments/taught_fit.yaml --seed 4 && slk data configs/experiments/taught_rows.yaml --seed 4
slk train configs/experiments/taught_fit.yaml --seed 5 && slk data configs/experiments/taught_rows.yaml --seed 5
slk train configs/experiments/taught_control_fit.yaml --seed 1 && slk data configs/experiments/taught_control_rows.yaml --seed 1
slk run configs/experiments/taught_student.yaml --seed 2
slk run configs/experiments/taught_student.yaml --seed 3
slk run configs/experiments/taught_student.yaml --seed 4
slk run configs/experiments/taught_student.yaml --seed 5
slk run configs/experiments/taught_control_student.yaml --seed 1
